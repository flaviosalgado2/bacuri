Como usar
# Subir o banco
docker compose up -d postgres

# Criar banco de testes (apenas para rodar testes)
docker exec postgres-dev psql -U nuxt_dev -d nuxt_dev -c "CREATE DATABASE nuxt_test;"

# Instalar dependências
npm install

# Aplicar migrations
npm run db:migrate

# Rodar em desenvolvimento
npm run dev

# Rodar testes
npm test
